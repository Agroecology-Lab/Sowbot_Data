 24    use_sim_time is the 'sim' LaunchConfiguration substitution (a string
 25    'true'/'false', not resolved until launch time) wrapped in ParameterValue

 27    'true'/'false'. It is NOT a plain Python bool — that would require
 28    knowing sim/real at generate_launch_description() parse time, which we
 29    don't: 'sim' itself is a LaunchConfiguration, only resolved at runtime.

 31    use_sim_time=True nodes simply sit waiting for a valid /clock until
 32    Gazebo is started via one of the UI buttons (Launch World/Spawn Robot or

 35    fake_nav2_server's own action servers must stop once that happens.
 36
 37    Replaces IncludeLaunchDescription of navigation_launch.py so we can
 38    omit collision_monitor, which requires observation_sources (a sensor)
 39    and throws InvalidParameterValueException on startup in Nav2 Jazzy when
 40    no sensor is configured — even with observation_sources: [] in the YAML.
 41    Re-add collision_monitor here once a lidar or ultrasonic is wired up.
 42
 43    Remappings mirror those in navigation_launch.py:
 44      cmd_vel -> cmd_vel_nav  (controller_server, behavior_server)
 45      cmd_vel -> cmd_vel_nav  (velocity_smoother input), cmd_vel_smoothed -> cmd_vel  (velocity_smoother output)
 46    tf/tf_static are remapped globally via the common remappings list.

 52        "parameters": [{'use_sim_time': use_sim_time_param}, params_file],
 53        "arguments": ['--ros-args', '--log-level', 'info'],
 54        "remappings": remappings,
 55    }

 57    return [
 58        Node(
 59            package='nav2_controller',

138def _kill_fake_nav2_action(sim_condition):
139    """Kill fake_nav2_server once real Nav2's lifecycle manager has actually
140    activated bt_navigator — both run a NavigateToPose action server and
141    would otherwise collide on the same action name.
142
143    fake_nav2_server itself keeps starting unconditionally in sim mode (see
144    the node below) because its wall-time /clock publisher is still needed:
145    real Nav2 launches at container boot here, well before Gazebo exists, so
146    something has to keep /clock and map->odom->base_link TF alive in the
147    meantime or Nav2's lifecycle/TF lookups fail outright. Once Gazebo's
148    ros_gz_bridge comes up (via any of the UI's Launch World/Spawn
149    Robot/Launch Sim buttons) its RELIABLE /clock wins by QoS precedence
150    automatically — but fake_nav2_server's ACTION SERVERS don't get
151    superseded that way, they just keep running and competing with
152    bt_navigator. So we kill the process outright, on a poll loop, the
153    moment bt_navigator is actually active.
154
155    BOUNDED, NOT INFINITE: bt_navigator activating depends on
156    lifecycle_manager_navigation's autostart bring-up succeeding, which is
157    exactly the kind of thing that's failed silently before in this codebase
158    (see the lifecycle-manager race this whole launch-file split was built
159    to fix). An earlier version of this watchdog looped `while true` with no
160    timeout — if bt_navigator never activates, that version would poll
161    forever with zero visible signal that anything was wrong, while
162    fake_nav2_server kept masking the failure by staying alive and looking
163    functional. This version caps at MAX_ATTEMPTS, logs progress every 10
164    attempts so it's visible in the boot log rather than silent, and on
165    timeout logs a loud, unambiguous ERROR and exits non-zero — making
166    "real Nav2 never came up" a visible failure instead of a quiet one.
167
168    Implemented as a plain bash poll loop rather than a custom node —
169    topological_nav_simulator (which owns fake_nav2_server) isn't a package
170    we maintain here, so adding a new executable to it isn't an option from
171    this launch file.
172    """
173    poll_interval_s = 2
174    max_attempts = 90   # 90 * 2s = 3 minutes — generous; Gazebo is NOT
175                        # required for bt_navigator to activate, only real
176                        # Nav2's own lifecycle bring-up, so this should
177                        # resolve quickly if it's going to resolve at all.
178    return ExecuteProcess(
179        condition=sim_condition,
180        cmd=[
181            '/bin/bash', '-c',
182            (
183                f'ATTEMPT=0; '
184                f'while [ "$ATTEMPT" -lt {max_attempts} ]; do '
185                f'sleep {poll_interval_s}; '
186                f'ATTEMPT=$((ATTEMPT + 1)); '
187                f'STATE=$(ros2 lifecycle get /bt_navigator 2>/dev/null); '
188                f'if [[ "$STATE" == active* ]]; then '
189                f'pkill -f fake_nav2_server || true; '
190                f'echo "[kill_fake_nav2] bt_navigator active after '
191                f'$((ATTEMPT * {poll_interval_s}))s — killed fake_nav2_server"; '
192                f'exit 0; '
193                f'fi; '
194                f'if [ $((ATTEMPT % 10)) -eq 0 ]; then '
195                f'echo "[kill_fake_nav2] still waiting for bt_navigator to '
196                f'activate ($((ATTEMPT * {poll_interval_s}))s elapsed, '
197                f'last state: ${{STATE:-unknown}})"; '
198                f'fi; '
199                f'done; '
200                f'echo "[kill_fake_nav2] ERROR: bt_navigator did not '
201                f'activate within $(({max_attempts} * {poll_interval_s}))s '
202                f'— real Nav2 lifecycle bring-up appears stuck. '
203                f'fake_nav2_server left running as a fallback. '
204                f'Check lifecycle_manager_navigation and bt_navigator logs." '
205                f'>&2; '
206                f'exit 1'
207            ),
208        ],
209        name='kill_fake_nav2_on_bt_active',
210        output='screen',
211    )

338def _map_origin():
339    """Return (lat, lon, yaw_deg) of the surveyed map origin, or None when it is not set.
340
341    MAP_ORIGIN_LAT and MAP_ORIGIN_LON (degrees) are the surveyed geodetic position of the
342    topological map's (0, 0). MAP_YAW_DEG (default 0) is the direction of the map x axis,
343    counter-clockwise from east. Both coordinates are needed: a half-set origin would
344    silently fall back to the wrong map, so raise instead.
345    """
346    raw = {name: os.getenv(name, '').strip() for name in ('MAP_ORIGIN_LAT', 'MAP_ORIGIN_LON')}
347    if not any(raw.values()):
348        return None
349    if not all(raw.values()):
350        raise RuntimeError('Set both MAP_ORIGIN_LAT and MAP_ORIGIN_LON, or neither.')
351    try:
352        lat, lon = float(raw['MAP_ORIGIN_LAT']), float(raw['MAP_ORIGIN_LON'])
353        yaw_deg = float(os.getenv('MAP_YAW_DEG', '').strip() or 0.0)
354    except ValueError as error:
355        raise RuntimeError(f'MAP_ORIGIN_LAT, MAP_ORIGIN_LON and MAP_YAW_DEG must be numbers: {error}') from error
356    if not (-90.0 <= lat <= 90.0 and -180.0 <= lon <= 180.0 and abs(yaw_deg) <= 360.0):
357        raise RuntimeError('MAP_ORIGIN_LAT must be within 90, MAP_ORIGIN_LON within 180 and MAP_YAW_DEG within 360.')
358    return lat, lon, yaw_deg

143            params_file,
144        ],
145    )

211def generate_launch_description():
212    """Build the simulation bringup with world generation, Gazebo, and navigation.
213

215    runs world generation. Gazebo, the /clock readiness gate and the map-to-odom
216    update are scheduled only if world generation exits zero. Nav2, its lifecycle

221    transform arrives on /tf; a timeout after 240 seconds or a failed wait leaves
222    them running.
223
224    Forward the camera launch arguments to the simulation: ``use_camera``
225    defaults to true, ``camera_width`` and ``camera_height`` to 320 and 240
226    pixels, and ``camera_rate`` to 10 Hz.

229    found in the ament index.

232    devkit_launch_pkg = get_package_share_directory('devkit_bringup')

234    world_arg = DeclareLaunchArgument(

243                    'Use sowbot_01.xacro (TrackedVehicle) or '
244                    'robo_caatinga.urdf.xacro (DiffDrive skid-steer).',
245    )
246    headless_arg = DeclareLaunchArgument(
247        'headless', default_value='false',
248        description='true: gz sim server-only, no GUI',
249    )
250    use_camera_arg = DeclareLaunchArgument(
251        'use_camera', default_value='true',
252        description='Enable the simulated camera sensor',
253    )
254    camera_width_arg = DeclareLaunchArgument(
255        'camera_width', default_value='320',
256        description='Simulated camera image width in pixels',
257    )
258    camera_height_arg = DeclareLaunchArgument(
259        'camera_height', default_value='240',
260        description='Simulated camera image height in pixels',
261    )
262    camera_rate_arg = DeclareLaunchArgument(
263        'camera_rate', default_value='10',
264        description='Simulated camera update rate in Hz',
265    )
266    x_arg = DeclareLaunchArgument('x', default_value='0.0')
267    y_arg = DeclareLaunchArgument('y', default_value='0.0')
268    z_arg = DeclareLaunchArgument('z', default_value='0.3')
269
270    # Ensure Forest3D-generated models are always findable by gz sim.
271    existing = os.environ.get('GZ_SIM_RESOURCE_PATH', '')
272    gz_resource_path = SetEnvironmentVariable(
273        'GZ_SIM_RESOURCE_PATH',
274        '/workspace/models'
275        ':/workspace/install/virtual_maize_field'
276        '/share/virtual_maize_field/models'
277        + (':' + existing if existing else ''),
278    )
279
280    # DEVKIT_URDF is read by sim.launch.py at generate_launch_description()
281    # time to resolve xacro_file_eager — launch_arguments can't be used there
282    # because substitutions aren't yet evaluated when os.path.join runs.
283    set_urdf_env = SetEnvironmentVariable('DEVKIT_URDF', LaunchConfiguration('urdf'))
284
285    # ── Gazebo + robot_state_publisher + spawn + ros_gz_bridge ───────────────
286    sim_launch = IncludeLaunchDescription(
287        PythonLaunchDescriptionSource(
288            os.path.join(pkg_agro, 'launch', 'sim.launch.py')
289        ),
290        launch_arguments={
291            'world': LaunchConfiguration('world'),
292            'urdf':  LaunchConfiguration('urdf'),
293            'headless': LaunchConfiguration('headless'),
294            'use_camera': LaunchConfiguration('use_camera'),
295            'camera_width': LaunchConfiguration('camera_width'),
296            'camera_height': LaunchConfiguration('camera_height'),
297            'camera_rate': LaunchConfiguration('camera_rate'),
298            'x':     LaunchConfiguration('x'),
299            'y':     LaunchConfiguration('y'),
300            'z':     LaunchConfiguration('z'),
301        }.items(),
302    )
303
304    # Kill any stale bridge from a previous session before sim_launch starts.
305    # Also kills fake_nav2_server (the boot-time /clock+TF stub started by
306    # sim_nav.launch.py) — real Nav2, started later in this file, takes over
307    # both the navigation backend and the /clock source (via ros_gz_bridge),
308    # so leaving fake_nav2_server alive would mean two NavigateToPose action
309    # servers competing for the same action name. See nav2_only.launch.py's
310    # module docstring for the full rationale (this mirrors that file's
311    # kill_fake_nav2 step for the other UI entry point).
312    # BUG FIXED HERE: pkill -f matches a process's FULL command line, not just
313    # the target program name. The old version's `pkill -f "gz sim"` ran
314    # inside `/bin/bash -c 'pkill -f "gz sim" || true; ...'` — and that bash
315    # invocation's own command line contains the literal substring "gz sim"
316    # (embedded in the script text passed to -c). So the very first pkill
317    # call matched its own parent shell and SIGTERM'd it before the other
318    # three pkill calls (ros_gz_sim, parameter_bridge, fake_nav2_server) ever
319    # ran. This is why preflight_pkill has died with exit code -15 on every
320    # single launch, every time — self-inflicted, not an external kill.
321    # Net effect: cleanup of stale processes from a prior (possibly
322    # SIGKILL'd) session never actually happened, which can leave a leftover
323    # gz sim / parameter_bridge / fake_nav2_server around to race or conflict
324    # with the new session (duplicate /clock-like publishers, stale DDS
325    # participants, etc.).
326    #
327    # Fix: bracket one character of each pattern (e.g. "[g]z sim" instead of
328    # "gz sim"). As a regex this still matches the literal string "gz sim" in
329    # a TARGET process's command line, but it no longer matches pkill's own
330    # invocation text (which contains "[g]z sim", not "gz sim").
331    preflight_pkill = ExecuteProcess(
332        cmd=['/bin/bash', '-c',
333             'pkill -f "[g]z sim" || true; '
334             'pkill -f "[r]os_gz_sim" || true; '
335             'pkill -f "[p]arameter_bridge" || true; '
336             'pkill -f "[f]ake_nav2_server" || true'],
337        name='preflight_pkill',
338        output='screen',
339    )
340
341    # sim_launch was previously a plain top-level action, started at the same
342    # instant as preflight_pkill (ros2 launch runs top-level actions
343    # concurrently, not in list order — the numbered docstring above was
344    # aspirational, not enforced). preflight_pkill's `pkill -f "gz sim"` then
345    # raced the freshly-started `gz sim -r ...` process from sim_launch and
346    # could kill it seconds after it started — gz_sim exits, spawn_robot
347    # (still in its "waiting for gz sim" loop) gets torn down with it, no
348    # entity ever spawns, and every TF/costmap error downstream (Nav2 seeing
349    # two unconnected trees, etc.) is fallout from that, not an independent
350    # bug. Gate sim_launch on preflight_pkill's exit so the kill always
351    # finishes before Gazebo starts.
352
353    # ── World generation (input-keyed, cached) ───────────────────────────────
354    # The Gazebo world is derived FROM the topo map by the Forest3D pipeline,
355    # and Gazebo reads it only at launch. The placement inputs are read from
356    # the "settings:" block of the last forest3d.yaml the UI wrote
357    # (plant_spacing, weed_density, etc.), defaulting to the pipeline's
358    # defaults when none exists yet, so Launch Sim replays the exact
359    # generation the user configured instead of silently resetting values.
360    # The world is cached: a hash of (resolved settings, topo map, crop/weed
361    # model files, uploaded terrain, and the generator script itself) is
362    # stored next to the world; if it still matches on the next press the
363    # expensive 40-60s regeneration is skipped and gz starts instantly. The
364    # key file lives beside the world in install/ (same container lifetime),
365    # so a fresh `manage.py up --sim` always regenerates once. sim_launch is
366    # gated on this step's exit so gz sim never races a half-written world.
367    # The logic lives in /workspace/worldgen.sh (mounted by manage.py) so the
368    # Launch Sim button and the UI Rebuild button share one implementation.
369    world_gen = ExecuteProcess(
370        cmd=['/bin/bash', '/workspace/worldgen.sh'],
371        name='world_gen',
372        output='screen',
373    )
374
375    start_sim_after_preflight = RegisterEventHandler(
376        OnProcessExit(
377            target_action=preflight_pkill,
378            on_exit=[world_gen],
379        )
380    )
381
382    def _launch_sim_if_worldgen_ok(event, context):
383        """Return the simulation launch on world generation success, or log its failure."""
384        if event.returncode != 0:
385            return [LogInfo(msg=(
386                f'[sowbot_sim] world_gen exited with code {event.returncode} — '
387                'not starting sim_launch (world may be missing/stale)'
388            ))]
389        return [sim_launch]
390
391    def _if_worldgen_ok(actions):
392        """Create an exit handler that gates actions on successful world generation."""
393        # Gate downstream actions on world_gen exiting 0. The failure is
394        # already logged by _launch_sim_if_worldgen_ok.
395        def _handler(event, context):
396            """Return the gated actions on success, or an empty list on failure."""
397            return list(actions) if event.returncode == 0 else []
398        return _handler
399
400    start_sim_after_worldgen = RegisterEventHandler(
401        OnProcessExit(
402            target_action=world_gen,
403            on_exit=_launch_sim_if_worldgen_ok,
404        )
405    )
406
407    # ── /clock readiness gate ────────────────────────────────────────────────
408    # Nav2, its lifecycle manager, fusioncore and the fusion relay all need
409    # /clock live (use_sim_time=True) and the robot spawned. They used to start
410    # on fixed timers (35s/40s) after world_gen, which raced on a slow machine
411    # (low RTF) and started the whole nav stack even when gz sim, the spawn or
412    # ros_gz_bridge had failed. ros_gz_bridge is only launched after the spawn
413    # step exits (see sim.launch.py), so /clock appearing means gz is up, the
414    # spawn step has finished and the bridge is forwarding.
415    #
416    # Single `ros2 topic echo --once` with the full timeout: one DDS
417    # participant, not many short-lived ones (see kill_bootstrap_tfs below).
418    # Exit non-zero on timeout so nothing downstream is released.
419    clock_gate = ExecuteProcess(
420        cmd=[
421            '/bin/bash', '-c',
422            'start_ts=$SECONDS; '
423            'if timeout 240 ros2 topic echo /clock --once >/dev/null 2>&1; then '
424            '  echo "[clock_gate] /clock live after $((SECONDS-start_ts))s"; '
425            'else '
426            '  echo "[clock_gate] ERROR: no /clock after $((SECONDS-start_ts))s '
427            '(gz sim, spawn or ros_gz_bridge did not come up) — '
428            'not starting Nav2/fusioncore"; exit 1; '
429            'fi',
430        ],
431        name='clock_gate',
432        output='screen',
433    )
434    start_clock_gate_after_worldgen = RegisterEventHandler(
435        OnProcessExit(target_action=world_gen, on_exit=_if_worldgen_ok([clock_gate])),
436    )
437
438    # Offsets below are measured from /clock first being seen, not from
439    # world_gen's exit. Same 5s spacing as before: server nodes (and
440    # fusioncore) first, lifecycle manager 5s later.
441    NAV_START_AFTER_CLOCK_S = 5.0
442    NAV_LIFECYCLE_AFTER_CLOCK_S = 10.0
443
444    # ── Nav2 (5s after /clock is live) ───────────────────────────────────────
445    # By the time Nav2 initialises, /clock is live and all TF frames carry
446    # Gazebo sim-time stamps — so use_sim_time=True works correctly from the
447    # start. Anchored on clock_gate's exit (see above), not a fixed sleep that
448    # needs retuning whenever spawn gets slower.
449    nav2_params = os.path.join(devkit_launch_pkg, 'config', 'nav2_params_sim.yaml')
450    nav2 = TimerAction(
451        period=NAV_START_AFTER_CLOCK_S,
452        actions=_nav2_sim_nodes(nav2_params, use_sim_time=True),
453    )
454    start_nav2_after_clock = RegisterEventHandler(
455        OnProcessExit(target_action=clock_gate, on_exit=_if_worldgen_ok([nav2])),
456    )
457
458    # ── Nav2 lifecycle manager (10s after /clock is live) ─────────────────────
459    # Started 5s AFTER the Nav2 server nodes above, not in the same batch.
460    # Previously this raced controller_server/behavior_server/bt_navigator's
461    # process startup, since ros2 launch forks everything in a TimerAction
462    # batch at the same instant with no guarantee the managed nodes' lifecycle
463    # services are actually up yet. Under load (gz sim RTF has run low on
464    # this stack) that race meant bt_navigator could be left configured but
465    # never activated, which is exactly the "[NAV2] Server unavailable" /
466    # STATUS_ABORTED failure topo nav goals were hitting. The 5s head start
467    # plus bond_timeout=15.0 (see _nav2_lifecycle_manager_node) gives every
468    # managed node real margin before lifecycle transitions are attempted.
469    nav2_lifecycle = TimerAction(
470        period=NAV_LIFECYCLE_AFTER_CLOCK_S,
471        actions=[_nav2_lifecycle_manager_node(nav2_params, use_sim_time=True)],
472    )
473    start_nav2_lifecycle_after_clock = RegisterEventHandler(
474        OnProcessExit(target_action=clock_gate, on_exit=_if_worldgen_ok([nav2_lifecycle])),
475    )
476
477    # Kill the wall-time bootstrap TF publishers from sim_nav.launch.py once
478    # a REAL odom->base_footprint source is actually live.  These were needed
479    # for topo nav localisation before Gazebo started, but after the bridge
480    # is live their wall-time stamps look ancient to Nav2's sim-time costmap
481    # (transform_tolerance=0.3s), causing "Costmap timed out waiting for
482    # update" and zero cmd_vel output.
483    #
484    # This was previously a fixed TimerAction(period=16.0), measured from
485    # generate_launch_description() being called — i.e. from before
486    # preflight_pkill even ran, NOT from when sim_launch (gz sim + spawn +
487    # bridge) actually started. spawn_robot/parameter_bridge routinely don't
488    # finish until t+35-40s (gz sim startup + 30s GUI-init wait in
489    # spawn_robot + bridge creation), so the fixed timer killed the bootstrap
490    # statics 20+ seconds before any real replacement existed. Result: a
491    # dead gap with NO odom->base_footprint publisher at all -> "Could not
492    # find a connection between 'odom' and 'base_footprint' ... two or more
493    # unconnected trees" (exactly the error seen in controller_server logs).
494    #
495    # Fix: poll for real /odom data (proof the DiffDrive bridge is actually
496    # forwarding gz Odometry, not just that the bridge node/topic exists)
497    # before killing anything. [STALE since DiffDrive publish_tf=false: the
498    # trigger is now fusioncore's own TF, see below.] base_footprint->base_link
499    # is covered by robot_state_publisher (fixed base_footprint_joint in the
500    # xacro). map->odom is left alone — it has no dynamic replacement.
501    #
502    # NOTE: this was previously a loop of `timeout 2 ros2 topic echo /odom
503    # --once` retried every 2s for up to 240s. That spawns a brand-new DDS
504    # participant on every single attempt, and gives each one only 2s to
505    # stand up, complete discovery against the bridge's existing publisher,
506    # match, and receive a sample -- a budget that's routinely too tight
507    # under loopback multicast discovery with ParticipantIndex=auto
508    # (successive short-lived participants can be slowed by TIME_WAIT churn
509    # on the discovery ports). Result: the check reported "no /odom data
510    # after 240s" and permanently killed the bootstrap statics even while
511    # /odom was being published continuously and reliably to every other
512    # node in the graph -- a false negative in the check, not an actual
513    # data outage. Fixed by using ONE participant with the full timeout
514    # budget instead of 120 short-lived ones.
515    # Trigger: a DYNAMIC odom->base_footprint (or ->base_link) transform on
516    # /tf. The bootstrap statics are on /tf_static (new-style args), so any
517    # /tf message with parent 'odom' can only come from fusioncore, the sole
518    # live source. Waiting on /odom instead (the previous trigger) proves
519    # nothing about fusioncore: /odom is gz ground truth, fusioncore starts on
520    # its own 35s timer and only publishes once it has initialised. Killing on
521    # /odom data therefore left a gap with no odom->base_footprint source.
522    # On timeout the statics are KEPT: killing them with no replacement is the
523    # "two or more unconnected trees" failure described above.
524    odom_tf_filter = (
525        "any(t.header.frame_id=='odom' and "
526        "t.child_frame_id in ('base_footprint','base_link') "
527        "for t in m.transforms)"
528    )
529    kill_bootstrap_tfs = ExecuteProcess(
530        cmd=[
531            '/bin/bash', '-c',
532            'start_ts=$SECONDS; timeout_s=240; '
533            f'if timeout "$timeout_s" ros2 topic echo /tf --once --filter "{odom_tf_filter}" >/dev/null 2>&1; then '
534            '  echo "[bootstrap_tf_killer] dynamic odom TF (fusioncore) live after $((SECONDS-start_ts))s"; '
535            '  pkill -f "static_transform_publishe[r].*__node:=odom_to_base_footprint_static" || true; '
536            '  pkill -f "static_transform_publishe[r].*__node:=base_footprint_to_base_link_static" || true; '
537            '  echo "[bootstrap_tf_killer] killed odom->base_footprint and base_footprint->base_link static publishers ($((SECONDS-start_ts))s elapsed)"; '
538            'else '
539            '  echo "[bootstrap_tf_killer] WARNING: no dynamic odom TF on /tf after $((SECONDS-start_ts))s — keeping bootstrap statics (fusioncore not publishing?)"; '
540            'fi',
541        ],
542        name='kill_bootstrap_tfs',
543        output='screen',
544    )
545
546    # ── Fix map->odom (after world-gen writes spawn_pose.txt) ─────────────────
547    # sim_nav.launch.py publishes map->odom at container boot, BEFORE
548    # spawn_pose.txt exists (worldgen only runs when the user presses Launch
549    # Sim), so _read_spawn_xy() falls back to identity (0,0,0). With identity,
550    # fusioncore's odom->base_footprint — whose odom frame is anchored at the
551    # robot's spawn point, so it sits near (0,0) at spawn — gets read as the
552    # robot's MAP-frame position. That is off by the entire spawn offset: the
553    # robot believes it is at map (0,0) while actually being at the spawn node
554    # (e.g. R1_IN at (0,-3.528)). Every goal, the costmap and the UI topo map
555    # are then wrong by that vector.
556    #
557    # worldgen writes spawn_pose.txt, so by the time world_gen exits we know
558    # the real spawn. Kill the stale boot-time static and republish map->odom
559    # with the spawn offset. tf2's static buffer keeps the first transform it
560    # saw for a frame pair, so the stale one MUST be killed before republishing
561    # — a second concurrent publisher for map->odom is ignored.
562    fix_map_to_odom = ExecuteProcess(
563        cmd=[
564            '/bin/bash', '-c',
565            'SPAWN_FILE=/workspace/spawn_pose.txt; '
566            'if [ ! -f "$SPAWN_FILE" ]; then '
567            '  echo "[map_to_odom_fixer] WARNING: $SPAWN_FILE missing — '
568            'leaving boot-time identity map->odom (alignment will be wrong)"; '
569            'else '
570            '  read -r SPAWN_X SPAWN_Y SPAWN_Z < "$SPAWN_FILE"; '
571            '  echo "[map_to_odom_fixer] spawn ($SPAWN_X, $SPAWN_Y) from $SPAWN_FILE"; '
572            '  echo "[map_to_odom_fixer] killing stale boot-time map->odom static"; '
573            '  pkill -f "static_transform_publishe[r].*__node:=map_to_odom_static" || true; '
574            '  sleep 1; '
575            '  exec ros2 run tf2_ros static_transform_publisher '
576            '    --x "$SPAWN_X" --y "$SPAWN_Y" --z 0 '
577            '    --qx 0 --qy 0 --qz 0 --qw 1 '
578            '    --frame-id map --child-frame-id odom '
579            '    --ros-args -r __node:=map_to_odom_fixed; '
580            'fi'
581        ],
582        name='map_to_odom_fixer',
583        output='screen',

316    def _move_to_pose(self, target_pose, goal_handle, action_label):

318        Simulate robot movement from current pose to target pose.
319

331        if not self._drive_to_position(target_pose, goal_handle, dt, action_label):

333
334        # Phase 3: Rotate to final orientation

340    def _rotate_toward_target(self, target_pose, goal_handle, dt):
341        """Rotate the robot to face the target position."""
342        dx = target_pose.position.x - self._robot.pose.position.x
343        dy = target_pose.position.y - self._robot.pose.position.y
344        dist = math.hypot(dx, dy)
345
346        if dist < self._goal_tol:
347            return True  # Already close, skip rotation
348
349        target_yaw = math.atan2(dy, dx)
350        current_yaw = _euler_from_quaternion(self._robot.pose.orientation)
351        yaw_diff = math.atan2(
352            math.sin(target_yaw - current_yaw),
353            math.cos(target_yaw - current_yaw),
354        )
355
356        if abs(yaw_diff) < 0.05:
357            return True  # Close enough
358
359        rotation_time = abs(yaw_diff) / self._ang_speed
360        steps = max(int(rotation_time / dt), 1)
361        start_yaw = current_yaw
362
363        for i in range(1, steps + 1):
364            if goal_handle.is_cancel_requested:
365                goal_handle.canceled()
366                return False
367            t = float(i) / steps
368            yaw = _slerp_yaw(start_yaw, target_yaw, t)
369            self._robot.pose.orientation = _quaternion_from_yaw(yaw)
370            time.sleep(dt)
371
372        return True
373
374    def _drive_to_position(self, target_pose, goal_handle, dt, action_label):
375        """Drive the robot in a straight line toward the target."""
376        while True:
377            if goal_handle.is_cancel_requested:
378                goal_handle.canceled()
379                return False
380
381            dx = target_pose.position.x - self._robot.pose.position.x
382            dy = target_pose.position.y - self._robot.pose.position.y
383            dist = math.hypot(dx, dy)
384
385            if dist < self._goal_tol:
386                # Snap to exact position
387                self._robot.pose.position.x = target_pose.position.x
388                self._robot.pose.position.y = target_pose.position.y
389                return True
390
391            # Move toward target
392            step = min(self._speed * dt, dist)
393            angle = math.atan2(dy, dx)
394            self._robot.pose.position.x += step * math.cos(angle)
395            self._robot.pose.position.y += step * math.sin(angle)
396            self._robot.pose.orientation = _quaternion_from_yaw(angle)
397
398            # Publish feedback
399            self._publish_feedback(goal_handle, action_label, dist)
400
401            time.sleep(dt)
402
403    def _rotate_to_orientation(self, target_pose, goal_handle, dt):
404        """Rotate the robot to the goal's final orientation."""
405        target_yaw = _euler_from_quaternion(target_pose.orientation)
406        current_yaw = _euler_from_quaternion(self._robot.pose.orientation)
407        yaw_diff = math.atan2(
408            math.sin(target_yaw - current_yaw),
409            math.cos(target_yaw - current_yaw),
410        )
411
412        if abs(yaw_diff) < 0.05:
413            self._robot.pose.orientation = target_pose.orientation
414            return True
415
416        rotation_time = abs(yaw_diff) / self._ang_speed
417        steps = max(int(rotation_time / dt), 1)
418        start_yaw = current_yaw
419
420        for i in range(1, steps + 1):
421            if goal_handle.is_cancel_requested:
422                goal_handle.canceled()
423                return False
424            t = float(i) / steps
425            yaw = _slerp_yaw(start_yaw, target_yaw, t)
426            self._robot.pose.orientation = _quaternion_from_yaw(yaw)
427            time.sleep(dt)
428
429        self._robot.pose.orientation = target_pose.orientation
430        return True

434        # NavigateToPose and NavigateThroughPoses have different feedback types

147        output="screen",
148        additional_env=gz_env_headless,
149        condition=IfCondition(LaunchConfiguration("headless")),
150    )
151
152    # ── 2. robot_state_publisher ──────────────────────────────────────────────
153    # Command([]) resolves PathJoinSubstitution correctly at node startup.
154    xacro_file = PathJoinSubstitution([pkg_share, "urdf", LaunchConfiguration("urdf")])
155    robot_state_publisher = Node(
156        package="robot_state_publisher",
157        executable="robot_state_publisher",
158        name="robot_state_publisher",
159        output="screen",
160        parameters=[{
161            "robot_description": ParameterValue(
162                Command([
163                    'xacro "', xacro_file,
164                    '" use_camera:=', LaunchConfiguration('use_camera'),
165                    ' camera_width:=', LaunchConfiguration('camera_width'),
166                    ' camera_height:=', LaunchConfiguration('camera_height'),
167                    ' camera_rate:=', LaunchConfiguration('camera_rate'),
168                ]), value_type=str
169            ),
170            "use_sim_time": True,
171            # Republish fixed joints on /tf at 50 Hz instead of one-shot
172            # /tf_static. Without this, Nav2 nodes that join after sim startup
173            # miss the transient-local /tf_static delivery (stamped at t=0 sim
174            # time, appears ancient by the time Nav2 is alive) and the
175            # base_footprint -> base_link transform is never seen.
176            "publish_frequency": 50.0,
177            "ignore_timestamp": True,
178        }],
179    )
180
181    # ── 3. Spawn robot ────────────────────────────────────────────────────────
182    # DEVKIT_URDF is set by sowbot_sim.launch.py via SetEnvironmentVariable
183    # before this subprocess starts, so the bash variable resolves at runtime
184    # to the model selected in the UI.  os.environ.get() is NOT used here
185    # because SetEnvironmentVariable is a substitution that hasn't evaluated
186    # yet when generate_launch_description() runs — it would always be empty.
187    urdf_dir = os.path.join(pkg_share, "urdf")
188    spawn_entity = ExecuteProcess(
189        cmd=[
190            "/bin/bash", "-c",
191            (
192                'echo "[spawn] waiting for gz sim..."; '
193                # Bounded wait: without a deadline a gz that never comes up
194                # leaves this process looping forever, so the spawn never exits
195                # and nothing downstream is released or reported. Exiting
196                # non-zero lets the bridge gate below fail closed.
197                'GZ_WAIT_S="${SPAWN_GZ_TIMEOUT_S:-180}"; '
198                'deadline=$((SECONDS + GZ_WAIT_S)); '
199                f'until {GZ_BIN} service -l 2>/dev/null | grep -q "/world/"; do '
200                'if (( SECONDS >= deadline )); then '
201                'echo "[spawn] gz sim reported no world within ${GZ_WAIT_S}s" >&2; '
202                'exit 1; fi; '
203                'sleep 2; done; '
204                'echo "[spawn] gz service ready — waiting 30s for GUI to finish init..."; '
205                'sleep 30; '
206                'echo "[spawn] spawning agro_robot"; '
207                f'_URDF="${{DEVKIT_URDF:-sowbot_01.xacro}}"; '
208                f'echo "[spawn] using URDF: {urdf_dir}/$_URDF"; '
209                # Spawn pose comes from topo_to_forest3d.py's HOME-node +
210                # terrain_offset calculation, not a hardcoded origin — the
211                # world's terrain/crops are shifted by terrain_offset, so a
212                # fixed (0,0) spawn drifts off the topo HOME position once
213                # that offset is non-zero. Falls back to legacy (0,0,0.01)
214                # if the file is missing (e.g. non-sim/manual runs).
215                'SPAWN_FILE="/workspace/spawn_pose.txt"; '
216                'if [ -f "$SPAWN_FILE" ]; then '
217                'read -r SPAWN_X SPAWN_Y SPAWN_Z < "$SPAWN_FILE"; '
218                'echo "[spawn] using spawn pose from $SPAWN_FILE: $SPAWN_X $SPAWN_Y $SPAWN_Z"; '
219                'else '
220                # 1.0m drop height, not a ground-level offset: terrain
221                # elevation at (0,0) varies with each generated world, so a
222                # near-zero Z (the old 0.01 default) spawns the robot
223                # partially embedded in non-flat terrain, tanking the physics
224                # real-time factor and producing bad wheel contact. Dropping
225                # from 1.0m and letting gravity settle onto the mesh is
226                # robust regardless of terrain shape at this fallback point.
227                'SPAWN_X=0.0; SPAWN_Y=0.0; SPAWN_Z=1.0; '
228                'echo "[spawn] WARNING: $SPAWN_FILE not found — falling back to (0,0,1.0), robot will drop onto terrain"; '
229                'fi; '
230                f'URDF=$(xacro "{urdf_dir}/$_URDF" '
231                '"use_camera:=$1" "camera_width:=$2" '
232                '"camera_height:=$3" "camera_rate:=$4") && '
233                'ros2 run ros_gz_sim create'
234                ' -name agro_robot'
235                ' -string "$URDF"'
236                ' -x "$SPAWN_X" -y "$SPAWN_Y" -z "$SPAWN_Z"'  # base_footprint on ground; base_link raised by wheel_radius via base_footprint_joint
237            ),
238            'spawn_robot',
239            LaunchConfiguration('use_camera'),
240            LaunchConfiguration('camera_width'),
241            LaunchConfiguration('camera_height'),
242            LaunchConfiguration('camera_rate'),
243        ],
244        name="spawn_robot",
245        output="screen",
246    )
247
248    # ── 4. Bridge (2 s after a successful spawn) ─────────────────────────────
249    # on_exit must be a callable, not a static list: a static list starts the
250    # bridge whatever the spawn status, which publishes /clock for a world with
251    # no robot and releases the Nav2/fusioncore clock gate.
252    bridge_config = os.path.join(pkg_share, "config", "ros_gz_bridge.yaml")
253
254    def _bridge_if_spawn_ok(event, context):
255        """Return the delayed bridge on spawn success, or a failure log otherwise."""
256        if event.returncode != 0:
257            return [LogInfo(msg=(
258                f"[sim] spawn_robot exited with code {event.returncode}: "
259                "not starting ros_gz_bridge"
260            ))]
261        return [
262            TimerAction(
263                period=2.0,
264                actions=[
265                    Node(
266                        package="ros_gz_bridge",
267                        executable="parameter_bridge",
268                        name="ros_gz_bridge",
269                        output="screen",
270                        respawn=True,
271                        respawn_delay=5.0,
272                        parameters=[
273                            {"use_sim_time": True},
274                            {"config_file": bridge_config},
275                        ],
276                    )
277                ],
278            )
279        ]
280
281    ros_gz_bridge = RegisterEventHandler(
282        OnProcessExit(target_action=spawn_entity, on_exit=_bridge_if_spawn_ok)
283    )
284
285    return LaunchDescription([
286        x_arg, y_arg, z_arg,
287        world_arg, urdf_arg, headless_arg,
288        use_camera_arg, camera_width_arg, camera_height_arg, camera_rate_arg,
289        gz_sim,
290        gz_sim_headless,
291        robot_state_publisher,
292        spawn_entity,
293        ros_gz_bridge,
294    ])
